   double a = 0, pp = 0, mm = 0, adx = 0;
   bool first = true, firstDx = true;
   for(int s = st; s >= sh; s--)
     {
      double h = Hi(tf, s), l = Lo(tf, s), pc = Cl(tf, s + 1), ph = Hi(tf, s + 1), pl = Lo(tf, s + 1);
      double tr = MathMax(h - l, MathMax(MathAbs(h - pc), MathAbs(l - pc)));
      double up = h - ph, dn = pl - l;
      double pdm = (up > dn && up > 0) ? up : 0, mdm = (dn > up && dn > 0) ? dn : 0;
      if(first) { a = tr; pp = pdm; mm = mdm; first = false; }
      else { a += (tr - a) / 14.0; pp += (pdm - pp) / 14.0; mm += (mdm - mm) / 14.0; }
      if(a <= 0) continue;
      double pdi = 100 * pp / a, mdi = 100 * mm / a;
      double dx = (pdi + mdi > 0) ? 100 * MathAbs(pdi - mdi) / (pdi + mdi) : 0;
      if(firstDx) { adx = dx; firstDx = false; }
      else adx += (dx - adx) / 14.0;
     }
   return adx;
  }

//--- failed breakout fade: the candle poked beyond the last 20-candle low / high and closed back inside
int TxFailFade(int tf)
  {
   int kh = iHighest(_Symbol, tf, MODE_HIGH, 20, 2), kl = iLowest(_Symbol, tf, MODE_LOW, 20, 2);
   if(kh < 0 || kl < 0) return 0;
   double hh = Hi(tf, kh), ll = Lo(tf, kl), o = Op(tf, 1), h = Hi(tf, 1), l = Lo(tf, 1), c = Cl(tf, 1);
   if(l < ll && c > ll && c > o) return 1;
   if(h > hh && c < hh && c < o) return -1;
   return 0;
  }

//--- Bollinger band re-entry: the candle before closed outside the band (20, 2) and this candle closed back inside
int TxBandBack(int tf)
  {
   double m1 = 0, m2 = 0;
   double s1 = TxSd(tf, 1, m1), s2 = TxSd(tf, 2, m2);
   double c1 = Cl(tf, 1), c2 = Cl(tf, 2);
   if(c2 < m2 - 2 * s2 && c1 > m1 - 2 * s1) return 1;
   if(c2 > m2 + 2 * s2 && c1 < m1 + 2 * s1) return -1;
   return 0;
  }

int TxPinBar(int tf)
  {
   double o = Op(tf, 1), h = Hi(tf, 1), l = Lo(tf, 1), c = Cl(tf, 1);
   double rng = h - l, body = MathAbs(c - o);
   if(rng <= 0) return 0;
   double lw = MathMin(o, c) - l, uw = h - MathMax(o, c), e20 = EMAv(tf, 20, 1);
   if(UpTr(tf, 1) && lw >= 2 * body && lw >= 0.6 * rng && l <= e20) return 1;
   if(DnTr(tf, 1) && uw >= 2 * body && uw >= 0.6 * rng && h >= e20) return -1;
   return 0;
  }

double TxMid(int tf, int n, int sh)
  {
   int kh = iHighest(_Symbol, tf, MODE_HIGH, n, sh), kl = iLowest(_Symbol, tf, MODE_LOW, n, sh);
   if(kh < 0 || kl < 0) return 0;
   return (Hi(tf, kh) + Lo(tf, kl)) / 2.0;
  }

//--- Ichimoku: Tenkan (9) crosses Kijun (26) with the close beyond the cloud (spans built 26 candles earlier)
int TxIchimoku(int tf)
  {
   if(iBars(_Symbol, tf) < 140) return 0;
   double t1 = TxMid(tf, 9, 1), k1 = TxMid(tf, 26, 1), t2 = TxMid(tf, 9, 2), k2 = TxMid(tf, 26, 2);
   double sa = (TxMid(tf, 9, 27) + TxMid(tf, 26, 27)) / 2.0, sb = TxMid(tf, 52, 27);
   double c = Cl(tf, 1);
   if(t1 > k1 && t2 <= k2 && c > MathMax(sa, sb)) return 1;
   if(t1 < k1 && t2 >= k2 && c < MathMin(sa, sb)) return -1;
   return 0;
  }

//--- inside day (yesterday inside the day before) and the candle closes beyond yesterday's high / low
int TxInsideDay(int tf)
  {
   if(iBars(_Symbol, PERIOD_D1) < 5) return 0;
   if(!(Hi(PERIOD_D1, 1) < Hi(PERIOD_D1, 2) && Lo(PERIOD_D1, 1) > Lo(PERIOD_D1, 2))) return 0;
   double ph = Hi(PERIOD_D1, 1), pl = Lo(PERIOD_D1, 1), c1 = Cl(tf, 1), c2 = Cl(tf, 2);
   if(c1 > ph && c2 <= ph) return 1;
   if(c1 < pl && c2 >= pl) return -1;
   return 0;
  }

bool TryTrendEntry(int code, int d, bool needTrend)
  {
   int m = M_TR;
   int tf = TxTf(code);
   int ly = (tf == PERIOD_H4) ? LY_T4 : LY_T1;
   int fm = 100 + code;
   string head = SideName(d) + " " + TxName(code) + " at " + IstClock(TimeCurrent()) + " IST";
   int hkey = HistKeyNow(HistSetup(ly, fm), d);
   if(needTrend && TrendSide() != d) { Note(m, head + " - no trade: the D1 / H4 trend does not agree", -1); return false; }
   if(g_scanDir[TfSlot(tf)] == d) { NoteMark(m, d, tf, fm, head + " - skipped: this " + TfName(tf) + " candle was already traded " + SideName(d) + " by another method", hkey); return false; }
   if(StackOnlyAfterBE && MethodTfAtRisk(m, tf)) { NoteMark(m, d, tf, fm, head + " - skipped: the open " + TfName(tf) + " Trend trade is not risk-free yet", hkey); return false; }
   string hv = "";
   if(HistoryMode == HIST_FOLLOW && HistSaysSkip(hkey, hv)) { NoteMark(m, d, tf, fm, head + " - skipped by history: " + hv, hkey); return false; }
   double atr = 0;
   double S = StopDistance(tf, d, atr);
   if(S <= 0) return false;
   RefreshRates();
   double spread = Ask - Bid;
   double entry = (d > 0) ? Ask : Bid;
   double sl = NormalizeDouble(entry - d * S, Digits);
   double lots = TxLots(tf, S);
   string gnote = "";
   lots = GuardLots(lots, d, gnote);
   if(lots <= 0) { NoteMark(m, d, tf, fm, head + " - skipped: " + gnote, hkey); return false; }
   string why = "";
   bool quiet = false;
   string fw = "";
   if(StudyFilter(d, S, fw)) { NoteMark(m, d, tf, fm, head + " - skipped: " + fw, hkey); return false; }
   if(UseSpreadFilter && MaxSpreadPctSL > 0 && spread > MaxSpreadPctSL / 100.0 * S) why = "spread is " + DoubleToString(100.0 * spread / S, 0) + "% of the stop (limit " + DoubleToString(MaxSpreadPctSL, 0) + "%)";
   else if(EntryGates(S, lots, why, quiet, m))
     {
      string tb = "";
      if(TimeBlock(tb)) why = tb;
     }
   if(why != "")
     {
      NoteMark(m, d, tf, fm, head + (quiet ? " - not taken: " : " - skipped: ") + why, hkey);
      if(!quiet) LogSkip(ly, fm, d, entry, sl, why);
      return false;
     }
   double xt = 0;
   g_nextXS = TxExit(code, xt);
   g_nextXT = xt;
   bool ok = PlaceTrade(ly, fm, d, sl, S, atr, lots, EX_TX, hkey);
   g_nextXS = -1;
   g_nextXT = 0;
   return ok;
  }

//--- new H1 / H4 candle: H4 entries first (they tested best), then H1; one new Trend trade per tick
void ScanTrend(bool newH1, bool newH4)
  {
   if(!UseTrendMethod || (!newH1 && !newH4)) return;
   bool nt = true;
   for(int pass = 0; pass < 2; pass++)
     {
      int tfp = (pass == 0) ? PERIOD_H4 : PERIOD_H1;
      if((tfp == PERIOD_H4 && !newH4) || (tfp == PERIOD_H1 && !newH1)) continue;
      for(int c = 1; c <= TX_N; c++)
        {
         if(TxTf(c) != tfp || !TxActive(c, nt)) continue;
         int d = TxSignal(c);
         if(d == 0) continue;
         if(TryTrendEntry(c, d, nt)) return;
        }
     }
  }

bool MemLoad(int tf)
  {
   int bars = iBars(_Symbol, tf) - 2;
   if(bars < 300) return false;
   int n = (bars > MNB) ? MNB : bars;
   for(int i = 0; i < n; i++)
     {
      int sh = n - i;
      g_wo[i] = iOpen(_Symbol, tf, sh);
      g_wh[i] = iHigh(_Symbol, tf, sh);
      g_wl[i] = iLow(_Symbol, tf, sh);
      g_wc[i] = iClose(_Symbol, tf, sh);
      g_wt[i] = iTime(_Symbol, tf, sh);
      if(g_wc[i] <= 0 || g_wh[i] <= 0 || g_wl[i] <= 0) return false;
     }
   g_wn = n;
   MemEmaAll();
   //--- ATR(14): simple average of the true range
   double tr[];
   ArrayResize(tr, n);
   double s = 0;
   for(int i = 0; i < n; i++)
     {
      tr[i] = (i == 0) ? g_wh[0] - g_wl[0] : MathMax(g_wh[i], g_wc[i - 1]) - MathMin(g_wl[i], g_wc[i - 1]);
      s += tr[i];
      if(i >= 14) s -= tr[i - 14];
      g_watr[i] = (i >= 13) ? s / 14.0 : 0.0;
     }
   //--- RSI(2), Wilder smoothing
   double up = 0, dn = 0, ar = 0.5;
   for(int i = 0; i < n; i++)
     {
      double d = (i == 0) ? 0.0 : g_wc[i] - g_wc[i - 1];
      double u = (d > 0) ? d : 0.0, w = (d < 0) ? -d : 0.0;
      if(i == 0) { up = u; dn = w; }
      else { up = ar * u + (1.0 - ar) * up; dn = ar * w + (1.0 - ar) * dn; }
      g_wrsi2[i] = 100.0 - 100.0 / (1.0 + up / MathMax(dn, 1e-12));
     }
   //--- Stochastic(14, 3, 3): %K from 3-candle sums, %D = 3-candle average of %K
   double hh[], ll[];
   ArrayResize(hh, n);
   ArrayResize(ll, n);
   for(int i = 0; i < n; i++)
     {
      hh[i] = 0;
      ll[i] = 0;
      g_wstk[i] = 50;
      g_wstd[i] = 50;
      if(i < 13) continue;
      double mx = g_wh[i], mn = g_wl[i];
      for(int k = i - 13; k < i; k++) { if(g_wh[k] > mx) mx = g_wh[k]; if(g_wl[k] < mn) mn = g_wl[k]; }
      hh[i] = mx;
      ll[i] = mn;
     }
   for(int i = 15; i < n; i++)
     {
      double num = 0, den = 0;
      for(int k = i - 2; k <= i; k++) { num += g_wc[k] - ll[k]; den += hh[k] - ll[k]; }
      g_wstk[i] = 100.0 * num / MathMax(den, 1e-12);
      if(i >= 17) g_wstd[i] = (g_wstk[i] + g_wstk[i - 1] + g_wstk[i - 2]) / 3.0;
     }
   //--- CCI(20) of the typical price, mean deviation
   for(int i = 0; i < n; i++)
     {
      g_wcci[i] = 0;
      if(i < 19) continue;
      double m = 0;
      for(int k = i - 19; k <= i; k++) m += (g_wh[k] + g_wl[k] + g_wc[k]) / 3.0;
      m /= 20.0;
      double md = 0;
      for(int k = i - 19; k <= i; k++) md += MathAbs((g_wh[k] + g_wl[k] + g_wc[k]) / 3.0 - m);
      md /= 20.0;
      g_wcci[i] = ((g_wh[i] + g_wl[i] + g_wc[i]) / 3.0 - m) / MathMax(0.015 * md, 1e-12);
     }
   //--- ADX(14), MT4 style
   double aa = 2.0 / 15.0, pdi = 0, mdi = 0, adx = 0;
   g_wadx[0] = 0;
   for(int i = 1; i < n; i++)
     {
      double pu = g_wh[i] - g_wh[i - 1], pd = g_wl[i - 1] - g_wl[i];
      if(pu < 0) pu = 0;
      if(pd < 0) pd = 0;
      if(pu > pd) pd = 0;
      else if(pu < pd) pu = 0;
      else { pu = 0; pd = 0; }
      double t = MathMax(g_wh[i] - g_wl[i], MathMax(MathAbs(g_wh[i] - g_wc[i - 1]), MathAbs(g_wl[i] - g_wc[i - 1])));
      double pp = (t > 0) ? 100.0 * pu / t : 0.0, qq = (t > 0) ? 100.0 * pd / t : 0.0;
      pdi = pp * aa + pdi * (1.0 - aa);
      mdi = qq * aa + mdi * (1.0 - aa);
      double sm = pdi + mdi;
      double dx = (sm > 0) ? 100.0 * MathAbs(pdi - mdi) / sm : 0.0;
      adx = dx * aa + adx * (1.0 - aa);
      g_wadx[i] = adx;
     }
   //--- latest confirmed 5-candle fractals (a fractal is known 2 candles after it formed)
   double lh = 0, lw = 0;
   for(int i = 0; i < n; i++)
     {
      g_wfh[i] = 0;
      g_wfl[i] = 0;
      if(i < 4) continue;
      int k = i - 2;
      if(g_wh[k] > g_wh[k - 1] && g_wh[k] > g_wh[k - 2] && g_wh[k] >= g_wh[k + 1] && g_wh[k] >= g_wh[k + 2]) lh = g_wh[k];
      if(g_wl[k] < g_wl[k - 1] && g_wl[k] < g_wl[k - 2] && g_wl[k] <= g_wl[k + 1] && g_wl[k] <= g_wl[k + 2]) lw = g_wl[k];
      g_wfh[i] = lh;
      g_wfl[i] = lw;
     }
   return true;
  }

//--- previous server day's high / low / close for the candle at index j
bool MemPrevDay(int j, double &ph, double &pl, double &pc)
  {
   int sh = iBarShift(_Symbol, PERIOD_D1, g_wt[j], false);
   if(sh < 0) return false;
   ph = iHigh(_Symbol, PERIOD_D1, sh + 1);
   pl = iLow(_Symbol, PERIOD_D1, sh + 1);
   pc = iClose(_Symbol, PERIOD_D1, sh + 1);
   return (ph > 0 && pl > 0 && pc > 0);
  }

//------------------------------------------------------------------
// THE 16 STRATEGIES of the learnt memory (decided at the close of candle j): +1 buy, -1 sell, 0 none
//------------------------------------------------------------------
int MemSignal(int sid, int j)
  {
   if(j < 20) return 0;
   double o = g_wo[j], h = g_wh[j], l = g_wl[j], c = g_wc[j];
   double po = g_wo[j - 1], pc = g_wc[j - 1];
   double atr = g_watr[j];
   if(atr <= 0) return 0;
   double rng = h - l, body = MathAbs(c - o);
   bool upT = (g_we20[j] > g_we50[j] && c > g_we200[j]);
   bool dnT = (g_we20[j] < g_we50[j] && c < g_we200[j]);
   bool L = false, S = false;
   switch(sid)
     {
      case 0:   // 3 bars in a row fade
         L = (c < pc && pc < g_wc[j - 2] && g_wc[j - 2] < g_wc[j - 3]);
         S = (c > pc && pc > g_wc[j - 2] && g_wc[j - 2] > g_wc[j - 3]);
         break;
      case 1:   // break of structure retest
        {
         bool brU = false, brD = false;
         double lvU = 0, lvD = 0;
         for(int i = j; i >= j - 9; i--)
           {
            if(g_wfh[i] > 0 && g_wc[i] > g_wfh[i]) brU = true;
            if(g_wfl[i] > 0 && g_wc[i] < g_wfl[i]) brD = true;
           }
         for(int i = j; i >= j - 10; i--)
           {
            if(lvU == 0 && g_wfh[i] > 0 && g_wc[i] > g_wfh[i] && g_wc[i - 1] <= g_wfh[i]) lvU = g_wfh[i];
            if(lvD == 0 && g_wfl[i] > 0 && g_wc[i] < g_wfl[i] && g_wc[i - 1] >= g_wfl[i]) lvD = g_wfl[i];
           }
         L = (brU && lvU > 0 && l <= lvU + 0.2 * atr && c > lvU && c > o);
         S = (brD && lvD > 0 && h >= lvD - 0.2 * atr && c < lvD && c < o);
         break;
        }
      case 2:   // CCI 200 fade
         L = (g_wcci[j] < -200);
         S = (g_wcci[j] > 200);
         break;
      case 3:   // EMA20 bounce with ADX > 25
         L = (upT && g_wadx[j] > 25 && l <= g_we20[j] && c > g_we20[j] && c > o);
         S = (dnT && g_wadx[j] > 25 && h >= g_we20[j] && c < g_we20[j] && c < o);
         break;
      case 4:   // fair value gap retest (gap of the last 10 candles)
        {
         bool gU = (l > g_wh[j - 2]), gD = (h < g_wl[j - 2]);
         int kU = -1, kD = -1;
         for(int k = j - 1; k >= j - 10 && k >= 2; k--)
           {
            if(kU < 0 && g_wl[k] > g_wh[k - 2]) kU = k;
            if(kD < 0 && g_wh[k] < g_wl[k - 2]) kD = k;
           }
         L = (!gU && kU >= 0 && g_we20[j] > g_we50[j] && l <= g_wl[kU] && c > g_wh[kU - 2] && c > o);
         S = (!gD && kD >= 0 && g_we20[j] < g_we50[j] && h >= g_wh[kD] && c < g_wl[kD - 2] && c < o);
         break;
        }
      case 5:   // IBS big-bar fade
        {
         double ibs = (c - l) / MathMax(rng, 1e-12);
         L = (rng > 1.5 * atr && ibs < 0.15);
         S = (rng > 1.5 * atr && ibs > 0.85);
         break;
        }
      case 6:   // Keltner fade
         L = (c < g_we20[j] - 2.0 * atr);
         S = (c > g_we20[j] + 2.0 * atr);
         break;
      case 7:   // pivot S1 / R1 bounce
        {
         double dh = 0, dl = 0, dc = 0;
         if(!MemPrevDay(j, dh, dl, dc)) return 0;
         double pv = (dh + dl + dc) / 3.0, r1 = 2.0 * pv - dl, s1 = 2.0 * pv - dh;
         L = (l <= s1 && c > s1);
         S = (h >= r1 && c < r1);
         break;
        }
      case 8:   // RSI(2) dip in the trend
         L = (upT && g_wrsi2[j] < 10);
         S = (dnT && g_wrsi2[j] > 90);
         break;
      case 9:   // round number rejection
        {
         double st = MemRoundStep();
         double fl0 = MathFloor(pc / st) * st, ce0 = MathCeil(pc / st) * st;
         L = (l <= fl0 && c > fl0);
         S = (h >= ce0 && c < ce0);
         break;
        }
      case 10:  // Stochastic 20 / 80 cross
         L = (g_wstk[j] > g_wstd[j] && g_wstk[j - 1] <= g_wstd[j - 1] && g_wstd[j] < 20);
         S = (g_wstk[j] < g_wstd[j] && g_wstk[j - 1] >= g_wstd[j - 1] && g_wstd[j] > 80);
         break;
      case 11:  // three soldiers / three crows
         L = (c > o && po < pc && g_wo[j - 2] < g_wc[j - 2] && c > pc && pc > g_wc[j - 2] && body > 0.5 * rng);
         S = (c < o && po > pc && g_wo[j - 2] > g_wc[j - 2] && c < pc && pc < g_wc[j - 2] && body > 0.5 * rng);
         break;
      case 12:  // z-score from EMA50 fade
         L = ((c - g_we50[j]) / atr < -3.0);
         S = ((c - g_we50[j]) / atr > 3.0);
         break;
      case 13:  // engulfing
         L = (c > o && pc < po && c >= po && o <= pc);
         S = (c < o && pc > po && c <= po && o >= pc);
         break;
      case 14:  // previous-day high / low sweep
        {
         double dh2 = 0, dl2 = 0, dc2 = 0;
         if(!MemPrevDay(j, dh2, dl2, dc2)) return 0;
         L = (l < dl2 && c > dl2);
         S = (h > dh2 && c < dh2);
         break;
        }
      case 15:  // round number break
        {
         double st2 = MemRoundStep();
         double a1 = MathFloor(c / st2), a0 = MathFloor(pc / st2);
         L = (a1 > a0);
         S = (a1 < a0);
         break;
        }
      default:
         return 0;
     }
   if(L && !S) return 1;
   if(S && !L) return -1;
   return 0;
  }

// SITUATION: higher-timeframe trends, volatility band, session, H4 W / M pattern
//------------------------------------------------------------------
//--- trend of the last completed candle of tf: +1 close > EMA50 and EMA20 > EMA50, -1 mirror, 0 flat; side200 = D1 close vs EMA200
int MemTrend(int tf, int &side200)
  {
   side200 = 0;
   int bars = iBars(_Symbol, tf) - 2;
   if(bars < 52) return 0;
   int n = (bars > MNB) ? MNB : bars;
   double a20 = 2.0 / 21.0, a50 = 2.0 / 51.0, a200 = 2.0 / 201.0;
   double e20 = 0, e50 = 0, e200 = 0;
   for(int i = 0; i < n; i++)
     {
      double c = iClose(_Symbol, tf, n - i);
      if(i == 0) { e20 = c; e50 = c; e200 = c; continue; }
      e20 = a20 * c + (1.0 - a20) * e20;
      e50 = a50 * c + (1.0 - a50) * e50;
      e200 = a200 * c + (1.0 - a200) * e200;
     }
   double cl = iClose(_Symbol, tf, 1);
   if(n >= 201) side200 = (cl > e200) ? 1 : -1;
   if(cl > e50 && e20 > e50) return 1;
   if(cl < e50 && e20 < e50) return -1;
   return 0;
  }

//--- W (+1) / M (-1) breakout on tf within the last 6 completed candles (0 none) - double bottom / top with a neckline break
int MemPattern(int tf)
  {
   int bars = iBars(_Symbol, tf) - 2;
   if(bars < 80) return 0;
   int n = (bars > MPB) ? MPB : bars;
   double h[], l[], c[], atr[];
   ArrayResize(h, n);
   ArrayResize(l, n);
   ArrayResize(c, n);
   ArrayResize(atr, n);
   for(int i = 0; i < n; i++)
     {
      int sh = n - i;
      h[i] = iHigh(_Symbol, tf, sh);
      l[i] = iLow(_Symbol, tf, sh);
      c[i] = iClose(_Symbol, tf, sh);
     }
   double tr[];
   ArrayResize(tr, n);
   double s = 0;
   for(int i = 0; i < n; i++)
     {
      tr[i] = (i == 0) ? h[0] - l[0] : MathMax(h[i], c[i - 1]) - MathMin(l[i], c[i - 1]);
      s += tr[i];
      if(i >= 14) s -= tr[i - 14];
      atr[i] = (i >= 13) ? s / 14.0 : 0.0;
     }
   int lo1 = -1, lo2 = -1, hi1 = -1, hi2 = -1, lastE = -1, lastS = 0;
   for(int j = 4; j < n; j++)
     {
      int k = j - 2;
      if(l[k] < l[k - 1] && l[k] < l[k - 2] && l[k] <= l[k + 1] && l[k] <= l[k + 2]) { lo1 = lo2; lo2 = k; }
      if(h[k] > h[k - 1] && h[k] > h[k - 2] && h[k] >= h[k + 1] && h[k] >= h[k + 2]) { hi1 = hi2; hi2 = k; }
      double a = atr[j];
      if(!(a > 0)) continue;
      if(lo1 >= 0 && lo2 - lo1 >= 5 && lo2 - lo1 <= 60 && MathAbs(l[lo2] - l[lo1]) <= 0.3 * a)
        {
         double neck = h[lo1];
         for(int q = lo1; q <= lo2; q++) if(h[q] > neck) neck = h[q];
         if(neck - MathMax(l[lo1], l[lo2]) >= a && c[j] > neck && c[j - 1] <= neck && j - lo2 <= 30) { lastE = j; lastS = 1; }
        }
      if(hi1 >= 0 && hi2 - hi1 >= 5 && hi2 - hi1 <= 60 && MathAbs(h[hi2] - h[hi1]) <= 0.3 * a)
        {
         double neck2 = l[hi1];
         for(int q = hi1; q <= hi2; q++) if(l[q] < neck2) neck2 = l[q];
         if(MathMin(h[hi1], h[hi2]) - neck2 >= a && c[j] < neck2 && c[j - 1] >= neck2 && j - hi2 <= 30) { lastE = j; lastS = -1; }
        }
     }
   if(lastE >= 0 && (n - 1) - lastE <= 5) return lastS;
   return 0;
  }

//--- session of the current time (UTC): 0 Asia 00-07, 1 London 07-12, 2 New York 12-17, 3 late 17-24, 4 crypto weekend
int MemSession()
  {
   datetime u = NowUTC();
   if(IsCrypto())
     {
      int dw = TimeDayOfWeek(u);
      if(dw == 0 || dw == 6) return 4;
     }
   int hr = TimeHour(u);
   if(hr < 7) return 0;
   if(hr < 12) return 1;
   if(hr < 17) return 2;
   return 3;
  }

//------------------------------------------------------------------
// LIVE LEARNING: every memory trade (real or shadow) updates its setup's live record; a failing setup is paused

string MemGV(int cell, string f) { return g_SP + "ML_" + _Symbol + "_C" + IntegerToString(cell) + "_" + f; }

void MemLiveAdd(int cell, double r, bool real)
  {
   if(cell < 0 || cell >= MEM_N) return;
   GVSet(MemGV(cell, "N"), GVGet(MemGV(cell, "N"), 0) + 1);
   GVSet(MemGV(cell, "R"), GVGet(MemGV(cell, "R"), 0) + r);
   if(r > 0) GVSet(MemGV(cell, "W"), GVGet(MemGV(cell, "W"), 0) + 1);
   if(real) GVSet(MemGV(cell, "NR"), GVGet(MemGV(cell, "NR"), 0) + 1);
  }

bool MemPaused(int cell)
  {
   if(cell < 0 || cell >= MEM_N || MemoryPauseAfter <= 0) return false;
   double n = GVGet(MemGV(cell, "N"), 0);
   if(n < MemoryPauseAfter) return false;
   return (GVGet(MemGV(cell, "R"), 0) / n <= MemoryPauseR);
  }

//--- totals of the live record (all setups of this chart)
void MemLiveTotals(int &n, int &nr, double &rs, int &paused)
  {
   n = 0;
   nr = 0;
   rs = 0;
   paused = 0;
   for(int i = 0; i < MEM_N; i++)
     {
      n += (int)GVGet(MemGV(i, "N"), 0);
      nr += (int)GVGet(MemGV(i, "NR"), 0);
      rs += GVGet(MemGV(i, "R"), 0);
      if(MemPaused(i)) paused++;
     }
  }

//------------------------------------------------------------------
// SHADOW TRADES (virtual: the EA follows the setup with the real prices but sends no order)
//------------------------------------------------------------------
void MemShadowAdd(int cell, int d, double slD, double tpD, int mex, int maxm, double atr)
  {
   for(int i = 0; i < MSHN; i++)
     {
      if(g_shOn[i]) continue;
      RefreshRates();
      double en = (d > 0) ? Ask : Bid;
      g_shOn[i] = true;
      g_shCell[i] = cell;
      g_shDir[i] = d;
      g_shEx[i] = mex;
      g_shMaxM[i] = maxm;
      g_shEn[i] = en;
      g_shR[i] = slD;
      g_shSL[i] = en - d * slD;
      g_shTP[i] = (tpD > 0) ? en + d * tpD : 0.0;
      g_shAtr[i] = atr;
      g_shPk[i] = 0;
      g_shT[i] = TimeCurrent();
      return;
     }
  }

void MemShadowClose(int i, double px, string why)
  {
   int d = g_shDir[i];
   double r = (g_shR[i] > 0) ? d * (px - g_shEn[i]) / g_shR[i] : 0.0;
   int cell = g_shCell[i];
   MemLiveAdd(cell, r, false);
   g_shOn[i] = false;
   if(cell >= 0 && cell < MEM_N)
      CsvLine("AK10_shadow_" + DateKey() + "_" + _Symbol + ".csv", "Date,Symbol,Side,OpenTimeIST,CloseTimeIST,Setup,Timeframe,Exit,Entry,ExitPrice,ResultR,Reason",
              DateKey() + "," + _Symbol + "," + SideName(d) + "," + IstStamp(g_shT[i]) + "," + IstStamp(TimeCurrent()) + "," + CsvSafe(g_msName[g_mSt[cell]]) + "," +
              ((g_mTf[cell] == 15) ? "M15" : "H1") + "," + MemExitName(g_shEx[i]) + "," + Px(g_shEn[i]) + "," + Px(px) + "," + DoubleToString(r, 2) + "," + why);
  }

void MemShadowTick()
  {
   RefreshRates();
   for(int i = 0; i < MSHN; i++)
     {
      if(!g_shOn[i]) continue;
      int d = g_shDir[i];
      double px = (d > 0) ? Bid : Ask;
      double fav = d * (px - g_shEn[i]);
      if(fav > g_shPk[i]) g_shPk[i] = fav;
      if(d * (px - g_shSL[i]) <= 0) { MemShadowClose(i, g_shSL[i], "stop"); continue; }
      if(g_shTP[i] > 0 && d * (px - g_shTP[i]) >= 0) { MemShadowClose(i, g_shTP[i], "target"); continue; }
      if(g_shEx[i] == MEX_TREND)
        {
         double a = g_shAtr[i];
         if(g_shPk[i] >= 2.0 * a && d * (g_shSL[i] - g_shEn[i]) < 0) g_shSL[i] = g_shEn[i] + d * 0.1 * a;
         if(g_shPk[i] >= 3.0 * a)
           {
            double ns = g_shEn[i] + d * (g_shPk[i] - 3.0 * a);
            if(d * (ns - g_shSL[i]) > 0) g_shSL[i] = ns;
           }
        }
      int sh = iBarShift(_Symbol, PERIOD_M1, g_shT[i], false);
      int mm = (sh >= 0) ? sh : (int)((TimeCurrent() - g_shT[i]) / 60);
      if(g_shMaxM[i] > 0 && mm >= g_shMaxM[i]) MemShadowClose(i, px, "time limit");
     }
  }

int MemShadowOpen()
  {
   int c = 0;
   for(int i = 0; i < MSHN; i++) if(g_shOn[i]) c++;
   return c;
  }

  {
   int c = 0;
   for(int i = OrdersTotal() - 1; i >= 0; i--)
     {
      if(!OrderSelect(i, SELECT_BY_POS, MODE_TRADES)) continue;
      if(!MyOrder() || MethodOfMagic(OrderMagicNumber()) != M_MEM) continue;
      long key = OrderKey();
      if(sid >= 0 && ((int)PGet(key, "MTF", 0) != tfm || (int)PGet(key, "FAM", -1) != sid)) continue;
      c++;
     }
   return c;
  }

//--- lot: fixed MemoryLot, or (risk lot mode) the memory risk % x setup strength (0.5x - 1.5x), never below the minimum lot
double MemLots(double slD, int cell)
  {
   double f = (g_cls == SC_GOLD && GoldMemoryMode == GM_SMALL) ? 0.5 : 1.0;      // gold Memory: small lot (option)
   if(LotMode != AK_LOT_RISK) return NormLot(((g_cls == SC_ETH) ? MemoryLotETH : MemoryLot) * f);
   double conf = (cell >= 0) ? g_mAvgR[cell] / 0.45 : 1.0;
   if(conf < 0.5) conf = 0.5;
   if(conf > 1.5) conf = 1.5;
   double v1 = ValuePerPrice(1.0), bal = AccountBalance();
   if(v1 <= 0 || bal <= 0 || slD <= 0) return NormLot(0);
   return NormLot(bal * MemoryRiskPct / 100.0 * conf * f / (slD * v1));
  }

bool PlaceMemTrade(int cell, int d, int tfm, int sit, double atr, double slD, double tpD, double lots, string sitTxt)
  {
   int cmd = (d > 0) ? OP_BUY : OP_SELL;
   int sid = g_mSt[cell], mex = g_mEx[cell];
   string cmt = TradeComment + " MEM " + ((tfm == 15) ? "M15" : "H1") + " S" + IntegerToString(sid) + " E" + IntegerToString(mex);
   int tk = -1, err = 0;

//--- a learnt setup fired in a learnt situation: real trade, shadow trade or skip
void MemTake(int cell, int d, int tfm, int sit, string sitTxt)
  {
   int sid = g_mSt[cell], mex = g_mEx[cell];
   double atr = g_watr[g_wn - 1];
   double slD = (mex == MEX_TREND) ? 1.5 * atr : atr;
   double tpD = (mex == MEX_TREND) ? 0.0 : (mex + 1) * atr;
   int maxm = MemMaxBars(mex) * tfm;
   string head = SideName(d) + " " + ((tfm == 15) ? "M15 " : "H1 ") + g_msName[sid] + " at " + IstClock(TimeCurrent()) + " IST (record " + IntegerToString(g_mN[cell]) + " times, won " +
                 DoubleToString(g_mWin[cell], 0) + "%, avg " + ((g_mAvgR[cell] >= 0) ? "+" : "") + DoubleToString(g_mAvgR[cell], 2) + "R)";
   g_memLastT = TimeCurrent();
   if(MemPaused(cell)) { g_memLast = head + " - skipped: this setup is paused (its live results fell below " + DoubleToString(MemoryPauseR, 2) + "R)"; return; }
   //--- gold: no entries in the hour around the daily rollover or late Friday (as in the analysis)
   int sh = TimeHour(TimeCurrent()), dw = TimeDayOfWeek(TimeCurrent());
   if(!IsCrypto() && (sh == 23 || sh == 0 || (dw == 5 && sh >= 22))) { g_memLast = head + " - skipped: rollover / Friday-close hour"; return; }
   if(MemOpenCount(tfm, sid) > 0 || MemShadowCount(tfm, sid) > 0) { g_memLast = head + " - skipped: this setup already has an open (or shadow) trade"; return; }
   if(!MemRealMode())
     {
      MemShadowAdd(cell, d, slD, tpD, mex, maxm, atr);
      g_memLast = head + " -> SHADOW trade (virtual, learning)";
      g_memAct = g_memLast;
      return;
     }
   if(g_memTick) { g_memLast = head + " - followed as a shadow trade (one new memory trade per candle)"; MemShadowAdd(cell, d, slD, tpD, mex, maxm, atr); g_memAct = g_memLast; return; }
   if(MemOpenCount(0, -1) >= MemoryMaxOpen) { g_memLast = head + " - skipped: " + IntegerToString(MemoryMaxOpen) + " memory trades already open (followed as a shadow trade)"; MemShadowAdd(cell, d, slD, tpD, mex, maxm, atr); g_memAct = g_memLast; return; }
   RefreshRates();
   double spread = Ask - Bid;
   if(UseSpreadFilter && MaxSpreadPctSL > 0 && spread > MaxSpreadPctSL / 100.0 * slD) { g_memLast = head + " - skipped: spread is " + DoubleToString(100.0 * spread / slD, 0) + "% of the stop"; return; }
   string tb = "";
   if(TimeBlock(tb)) { g_memLast = head + " - skipped: " + tb; return; }
   double lots = MemLots(slD, cell);
   string gnote = "";
   lots = GuardLots(lots, d, gnote);
   if(lots <= 0) { g_memLast = head + " - skipped: " + gnote; return; }
   string why = "";
   bool quiet = false;
   if(!EntryGates(slD, lots, why, quiet, M_MEM))
     {
      g_memLast = head + " - not taken: " + why + (quiet ? " (followed as a shadow trade)" : "");
      if(quiet) { MemShadowAdd(cell, d, slD, tpD, mex, maxm, atr); g_memAct = g_memLast; }
      return;
     }
   PlaceMemTrade(cell, d, tfm, sit, atr, slD, tpD, lots, sitTxt);
  }

//--- a new M15 or H1 candle closed: run the 16 strategies, look each signal up in the memory
void MemCheckTf(int tf)
  {
   int tfm = (tf == PERIOD_M15) ? 15 : 60;
   int s200 = 0, dummy = 0;
   int t1 = 0, t2 = 0, t3 = 0;
   if(tfm == 15) { t1 = MemTrend(PERIOD_H1, dummy); t2 = MemTrend(PERIOD_H4, dummy); t3 = MemTrend(PERIOD_D1, dummy); }
   else { t1 = MemTrend(PERIOD_H4, dummy); t2 = MemTrend(PERIOD_D1, s200); t3 = s200; }
   int vb = VolBand(), ss = MemSession(), pat = MemPattern(PERIOD_H4);
   string sitTxt = MemSitText(tfm, t1, t2, t3, vb, ss, pat);
   g_memSit = ((tfm == 15) ? "M15 signals: " : "H1 signals: ") + sitTxt;
   if(!MemLoad(tf)) return;
   int j = g_wn - 1;
   for(int sid = 0; sid < MEM_STRATS; sid++)
     {
      int d = MemSignal(sid, j);
      if(d == 0) continue;
      int A = ((t1 == d) ? 1 : 0) + ((t2 == d) ? 1 : 0) + ((t3 == d) ? 1 : 0);
      int P = (pat == d) ? 2 : ((pat == 0) ? 1 : 0);
      int sit = ((A * 3 + vb) * 5 + ss) * 3 + P;
      int cell = MemFindCell(tfm, sid, sit);
      if(cell < 0)
        {
         int bc = MemBestCellOf(tfm, sid);
         if(bc >= 0)
           {
            g_memLast = SideName(d) + " " + ((tfm == 15) ? "M15 " : "H1 ") + g_msName[sid] + " at " + IstClock(TimeCurrent()) + " IST - no trade: in THIS situation (" + IntegerToString(A) +
                        " of 3 trends agree) it was not profitable after spread (learnt only in other situations)";
            g_memLastT = TimeCurrent();
           }
         continue;
        }
      MemTake(cell, d, tfm, sit, sitTxt);
     }
  }

void MemScan(bool newM15, bool newH1)
  {
   g_memTick = false;
   if(MemoryMode == MEM_OFF || !MemTradesHere()) return;
   if(g_cls == SC_GOLD && GoldMemoryMode == GM_OFF) return;                      // gold Memory switched off (option)
   if(newH1) MemCheckTf(PERIOD_H1);

  }

//--- management of a memory trade (called from ManageOne): learnt exit, time limit, tiers for the cards
void MemManageOne(int tk, long key, int dir, double open, double sl, double px, double R, double fav, double peak, int mm)
  {
   int mex = (int)PGet(key, "MEX", MEX_TREND);
   double atr = PGet(key, "ATR", 0);
   if(atr <= 0) atr = (mex == MEX_TREND) ? R / 1.5 : R;
   int maxm = (int)PGet(key, "MAXM", 0);
   if(maxm > 0 && mm >= maxm) { CloseTrade(tk, dir, key, CR_TIME); return; }
   if(mex != MEX_TREND)
     {
      double tpx = PGet(key, "TP", 0);
      if(tpx > 0 && OrderSelect(tk, SELECT_BY_TICKET) && OrderTakeProfit() <= 0)
        {
         if(dir * (px - tpx) >= 0) { CloseTrade(tk, dir, key, CR_TP); return; }
         SetTP(tk, dir, tpx);
        }
     }
   else
     {
      //--- TREND RIDE: stop to entry + 0.1 ATR at +2 ATR, trailing 3 ATR behind the best price from +3 ATR
      if(peak >= 2.0 * atr && (sl <= 0 || dir * (sl - open) < 0))
        {
         if(ModifySL(tk, dir, key, open + dir * 0.1 * atr)) PSet(key, "NLN", 1);
         else if(fav <= 0.1 * atr) { CloseTrade(tk, dir, key, CR_NOLOSS); return; }
         if(!OrderSelect(tk, SELECT_BY_TICKET)) return;
         sl = OrderStopLoss();
        }
      if(peak >= 3.0 * atr)
        {
         double ns = open + dir * (peak - 3.0 * atr);
         if(sl <= 0 || dir * (ns - sl) > MathMax(Point, 0.05 * atr))
           {
            if(ModifySL(tk, dir, key, ns)) PSet(key, "TRL", 1);
           }
        }
     }
   int tier = TierOfR((R > 0) ? peak / R : 0);
   int told = (int)PGet(key, "TIER", 0);
   if(tier > told)
     {
      PSet(key, "TIER", tier);
      if(tier >= TierCardFrom && MemoryCards) NotifyTier(key, dir, tier);
     }
  }

             " USD per 0.1 lot -> a $10 scalp averages " + Usd(ev) + " USD -> " + ((ev > 0) ? "edge" : "no M1/M5 scalp (spread eats it)");
  }

string HtmlEsc(string s) { StringReplace(s, "&", "&amp;"); StringReplace(s, "<", "&lt;"); StringReplace(s, ">", "&gt;"); return s; }
string RepeatCh(int cp, int n) { string d = ""; for(int i = 0; i < n; i++) d += ShortToString((ushort)cp); return d; }

int HexToInt(string h)
  {
   int v = 0;
   int n = StringLen(h);
   for(int i = 0; i < n; i++)
     {
      ushort c = StringGetCharacter(h, i);
      int d = 0;
      if(c >= '0' && c <= '9') d = c - '0';
      else if(c >= 'a' && c <= 'f') d = 10 + c - 'a';
      else if(c >= 'A' && c <= 'F') d = 10 + c - 'A';
      v = v * 16 + d;
     }
   return v;
  }

string JsonEscape(string s)
  {
   string r = "";
   int n = StringLen(s);
   for(int i = 0; i < n; i++)
     {
      ushort c = StringGetCharacter(s, i);
      if(c == '\\') r += "\\\\";
      else if(c == '"') r += "\\\"";
      else if(c == '\n') r += "\\n";
      else if(c == '\r') r += "\\r";
      else if(c == '\t') r += "\\t";
      else if(c < 32) r += " ";
      else r += ShortToString(c);
     }
   return r;
  }

string JsonValue(string src, string key, int from)
  {
   if(from < 0) return "";
   string pat = "\"" + key + "\":";
   int p = StringFind(src, pat, from);
   if(p < 0) return "";
   p += StringLen(pat);
   int n = StringLen(src);
   while(p < n && StringGetCharacter(src, p) == ' ') p++;
   if(p >= n) return "";
   if(StringGetCharacter(src, p) == '"')
     {
      p++;
      string out = "";
      while(p < n)
        {
         ushort c = StringGetCharacter(src, p);
         if(c == '\\' && p + 1 < n)
           {
            ushort e = StringGetCharacter(src, p + 1);
            if(e == 'u' && p + 5 < n) { out += ShortToString((ushort)HexToInt(StringSubstr(src, p + 2, 4))); p += 6; continue; }
            if(e == 'n') out += "\n";
            else if(e == 't') out += " ";
            else out += ShortToString(e);
            p += 2;
            continue;
           }
         if(c == '"') break;
         out += ShortToString(c);
         p++;
        }
      return out;
     }
   int q = p;
   while(q < n)
     {
      ushort c2 = StringGetCharacter(src, q);
      if(c2 == ',' || c2 == '}' || c2 == ']') break;
      q++;
     }
   return TrimStr(StringSubstr(src, p, q - p));
  }

string StripHtml(string t)
  {
   string r = "";
   bool inTag = false;
   int n = StringLen(t);
   for(int i = 0; i < n; i++)
     {
      ushort c = StringGetCharacter(t, i);
      if(c == '<') { inTag = true; continue; }
      if(c == '>' && inTag) { inTag = false; continue; }
      if(!inTag) r += ShortToString(c);
     }
   StringReplace(r, "&lt;", "<");
   StringReplace(r, "&gt;", ">");
   StringReplace(r, "&amp;", "&");
   return r;
  }

int HttpRequest(string method, string url, string body, string &response)
  {
   char data[];
   char res[];
   string rh = "";
   string headers = "Content-Type: application/json\r\n";
   if(StringLen(body) > 0)
     {
      int n = StringToCharArray(body, data, 0, WHOLE_ARRAY, CP_UTF8);
      if(n > 0) ArrayResize(data, n - 1);
     }
   else ArrayResize(data, 0);
   ResetLastError();
   int code = WebRequest(method, url, headers, 5000, data, res, rh);
   if(code == -1)
     {
      int e = GetLastError();
      response = "MT4 error " + IntegerToString(e);
      if(e == 4060 || e == 4014)
        {
         if(!g_tgUrlErr) Print("[TG] WebRequest blocked. Add https://api.telegram.org in Tools > Options > Expert Advisors > Allow WebRequest for listed URL.");
         g_tgUrlErr = true;
        }
      return -1;
     }
   response = CharArrayToString(res, 0, WHOLE_ARRAY, CP_UTF8);
   return code;
  }

//--- send a text message; returns the Telegram message id (> 0) or -1. markup = inline keyboard JSON ("" none), replyTo = message id (0 none)
int TgSendEx(string chat, string text, string markup, int replyTo)
  {
   if(StringLen(chat) == 0) return -1;
   if(TgPreviewOn()) return TgPreview(chat, "MSG", text, markup, replyTo);
   if(!g_tg || g_tgUrlErr) return -1;
   string tail = (StringLen(markup) > 0) ? ",\"reply_markup\":" + markup : "";
   string rep = (replyTo > 0) ? ",\"reply_parameters\":{\"message_id\":" + IntegerToString(replyTo) + ",\"allow_sending_without_reply\":true}" : "";
   string body = "{\"chat_id\":\"" + JsonEscape(chat) + "\",\"text\":\"" + JsonEscape(text) + "\",\"disable_web_page_preview\":true,\"parse_mode\":\"HTML\"" + rep + tail + "}";
   string resp = "";
   int code = HttpRequest("POST", TgApi("sendMessage"), body, resp);
   if(code == 400 && replyTo > 0)
     {
      //--- the message to reply to may be gone: send without the reply
      body = "{\"chat_id\":\"" + JsonEscape(chat) + "\",\"text\":\"" + JsonEscape(text) + "\",\"disable_web_page_preview\":true,\"parse_mode\":\"HTML\"" + tail + "}";
      code = HttpRequest("POST", TgApi("sendMessage"), body, resp);
     }
   if(code == 400)
     {
      //--- formatting rejected: the same text without formatting, so the message is never lost
      body = "{\"chat_id\":\"" + JsonEscape(chat) + "\",\"text\":\"" + JsonEscape(StripHtml(text)) + "\",\"disable_web_page_preview\":true" + tail + "}";
      code = HttpRequest("POST", TgApi("sendMessage"), body, resp);
     }
   if(code == 200 && StringFind(resp, "\"ok\":true") >= 0)
     {
      int mp = StringFind(resp, "\"message_id\":");
      int id = (mp >= 0) ? (int)StringToInteger(JsonValue(resp, "message_id", mp)) : 1;
      return (id > 0) ? id : 1;
     }
   Print("[TG] send to ", ChatLabel(chat), " failed: HTTP ", code, " ", StringSubstr(resp, 0, 160));
   return -1;
  }

string HashTag(string s)
  {
   string t = "";
   int n = StringLen(s);
   for(int i = 0; i < n; i++)
     {
      ushort ch = StringGetCharacter(s, i);
      if((ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z') || (ch >= '0' && ch <= '9')) t += ShortToString(ch);
     }
   return "#" + t;
  }

datetime g_swBar = 0;

//--- an alert only when the entry could really be taken now: the same gates as the entry itself (no side effects)
bool AlertGatesOK(int m, int ly, int d)
  {
   int tf = LayerTf(ly);
   int ex = EX_TREND;
   string w = "";
   if(!MarketPlan(d, ly, ex, w)) return false;
   if(!InSession()) return false;
   if(StackOnlyAfterBE && MethodTfAtRisk(m, tf)) return false;
   double atr = 0;
   double S = StopDistance(tf, d, atr);
   if(S <= 0) return false;
   double lots = TradeLots(S);
   if(m == M_PB && g_cls == SC_GOLD && GoldPullbackMode == GM_SMALL) lots = NormLot(lots * 0.5);
   if(StudyFilter(d, S, w)) return false;
   RefreshRates();
   if(UseSpreadFilter && MaxSpreadPctSL > 0 && (Ask - Bid) > MaxSpreadPctSL / 100.0 * S) return false;
   bool quiet = false;
   if(!EntryGates(S, lots, w, quiet, m)) return false;
   string tb = "";
   if(TimeBlock(tb)) return false;
   return true;
  }

void SetupWatchTick()
  {
   datetime b = iTime(_Symbol, PERIOD_M1, 0);
   if(b == g_swBar) return;
   g_swBar = b;
   g_swTxt = "";
   if(!SymbolSupported() || SymbolPaused()) return;
   int ts = TrendSide();
   double a1 = ATR(PERIOD_H1, 1);
   if(a1 <= 0 || iBars(_Symbol, PERIOD_H1) < 120) return;
   RefreshRates();
   double px = Bid;
   int kind = 0, dir = 0;
   double lvl = 0;
   string need = "";
   int lv = (SetupAlertLevel < 1) ? 1 : ((SetupAlertLevel > 3) ? 3 : SetupAlertLevel);
   double o1 = iOpen(_Symbol, PERIOD_H1, 0);
   bool withCandle = (ts != 0 && ts * (px - o1) > 0);                  // the running H1 candle already moves with the trend
   //--- 1. breakout watch: D1 + H4 + H1 agree, price close to the H1 20-candle high / low (level 1: 0.25 ATR, 2: 0.2 ATR and
   //---    moving towards it, 3: 0.1 ATR) and the entry could really be taken now
   if(ts != 0 && UseBreakout && BreakoutH1 && V10Here() && TfAgrees(PERIOD_H1, ts) && TfAgrees(PERIOD_H4, ts))
     {
      int k = (ts > 0) ? iHighest(_Symbol, PERIOD_H1, MODE_HIGH, 20, 1) : iLowest(_Symbol, PERIOD_H1, MODE_LOW, 20, 1);
      if(k >= 0)
        {
         double L = (ts > 0) ? Hi(PERIOD_H1, k) : Lo(PERIOD_H1, k);
         double dist = ts * (L - px);
         double maxD = (lv == 1) ? 0.25 : ((lv == 2) ? 0.2 : 0.1);
         if(dist >= 0 && dist <= maxD * a1 && (lv == 1 || withCandle) &&
            (AlertGatesOK(M_BO, LY_H1X, ts) || (BreakoutM15 && AlertGatesOK(M_BO, LY_M15X, ts))))
           { kind = 1; dir = ts; lvl = L; need = "a strong 1-hour close " + ((ts > 0) ? "above" : "below") + " this level"; }
        }
     }
   //--- 2. pull-back watch: H1 trend, price back in the EMA20 zone (level 2: and turning back with the trend; level 3: and
   //---    already past the last candle's high / low) and the entry could really be taken now
   if(kind == 0 && ts != 0 && UsePullback && PullbackH1 && V10Here() && !(g_cls == SC_GOLD && GoldPullbackMode == GM_OFF))
     {
      double e20 = EMAv(PERIOD_H1, 20, 0), e50 = EMAv(PERIOD_H1, 50, 0);
      bool trendH1 = (ts > 0) ? (e20 > e50) : (e20 < e50);
      bool zone = (MathAbs(px - e20) <= 0.15 * a1);
      if(lv >= 2) zone = zone || ((ts > 0) ? (Lo(PERIOD_H1, 1) <= EMAv(PERIOD_H1, 20, 1) + 0.15 * a1) : (Hi(PERIOD_H1, 1) >= EMAv(PERIOD_H1, 20, 1) - 0.15 * a1));
      bool turn = (lv == 1) || (withCandle && ts * (px - e20) > 0 && (lv == 2 || ts * (px - ((ts > 0) ? Hi(PERIOD_H1, 1) : Lo(PERIOD_H1, 1))) > 0));
      if(trendH1 && zone && turn && ts * (px - e50) > 0 &&
         (AlertGatesOK(M_PB, LY_H1, ts) || (PullbackM15 && AlertGatesOK(M_PB, LY_M15, ts))))
        { kind = 2; dir = ts; lvl = e20; need = "a strong 1-hour candle back " + ((ts > 0) ? "up" : "down") + " from this zone"; }
     }
   //--- (no alert for the ETH range entries: in the test only 2% of those alerts became a trade)
   if(kind == 0) return;
   string what = (kind == 1) ? ((dir > 0) ? "breakout above " : "breakdown below ") : "pull-back into the trend zone near ";
   string iks = (TgPreviewOn() ? "AK10QIK_" : "AK10IK_") + _Symbol + "_" + IntegerToString(dir) + "_" + IntegerToString(kind);
   if(TimeCurrent() - (datetime)GVGet(iks, 0) >= 4 * 3600) { GVSet(iks, (double)TimeCurrent()); InstaSetup(dir, what, lvl, need); }
   g_swTxt = SideName(dir) + " " + ((kind == 1) ? ((dir > 0) ? "breakout above " : "breakdown below ") : "pull-back zone ") + Px(lvl);   // panel (short)
   if(!SendSetupAlerts || (!g_tg && !TgPreviewOn())) return;
   //--- limits: once per symbol + side + kind every 4 hours; daily caps; never right after another post
   string sk = (TgPreviewOn() ? "AK10QK_" : "AK10K_") + _Symbol + "_" + IntegerToString(dir) + "_" + IntegerToString(kind);
   if(TimeCurrent() - (datetime)GVGet(sk, 0) < 4 * 3600) return;
   if(CountMyOpen(-1) >= MaxTradesPerSymbol) return;
   string dk = DateKey();
   bool pubOk = SetupAlertsPublic && SendToPublic && ContentHoursNow() && GVGet(TgGV("SWP" + dk), 0) < SetupAlertsPerDay && ContentSlotFree(true);
   bool vipOk = SendToVip && GVGet(TgGV("SWV" + dk), 0) < 2 * SetupAlertsPerDay;
   if(!pubOk && !vipOk) return;
   GVSet(sk, (double)TimeCurrent());
   GVSet(WatchGV(dir), (double)TimeCurrent());
   string wk = WatchKey(dir);
   string head = Emo(0x1F440) + " <b>SETUP LOADING...</b>  " + HashTag(_Symbol) + "\n" + Divider() +
                 "\n" + SideDot(dir) + " Bias: <b>" + SideName(dir) + "</b>  " + ShortToString((ushort)0x2022) + "  " + SymNice(_Symbol) + " " + SymEmoji(_Symbol) +
                 "\n" + Emo(0x1F4CD) + " Watching: <b>" + what + "</b><code>" + Px(lvl) + "</code>" +
                 "\n" + Emo(0x23F3) + " Needs: " + need +
                 "\n" + TrendLine(dir) + "\n" + Divider();
   string vip = head + "\n" + Emo(0x1F514) + " If it confirms, the full signal drops right here. No confirmation = no trade." + Sig();
   string pub = head + "\n" + Emo(0x1F514) + " Notifications ON - the entry comes the moment it confirms." +
                "\n" + Emo(0x1F48E) + " VIP gets the full plan instantly (stop, targets, live management)." + Sig();
   if(vipOk && SendOnceId(wk, VipChannelId, vip, "", 0) > 0) TgInc("SWV" + dk);
   if(pubOk && SendOnceId(wk, PublicChannelId, pub, VipButtons(), 0) > 0) { TgInc("SWP" + dk); ContentMark(); }
   SendOnceId(wk, AdminChatId, "[setup alert] " + _Symbol + " " + SideName(dir) + " " + what + Px(lvl) + " - needs " + need, "", 0);
  }

// CHART PHOTOS (entry and exit): the panel is hidden, the boxes stay in the picture

//--- price range of the photo: the candles it shows + every level of the open trades + up to two extra prices (entry / exit
//--- of a trade that just closed), so the whole trade is always in the picture
bool PhotoRange(double x1, double x2, double &hi, double &lo)
  {
   int tf = Period();
   int sc = (int)ChartGetInteger(0, CHART_SCALE, 0);
   if(sc < 0) sc = 0;
   if(sc > 5) sc = 5;
   double shp = ChartGetDouble(0, CHART_SHIFT_SIZE, 0);
   if(shp < 10 || shp > 50) shp = 20;
   int nb = (int)(ChartPhotoWidth * (1.0 - shp / 100.0)) / (1 << sc);
   if(nb > iBars(_Symbol, tf) - 1) nb = iBars(_Symbol, tf) - 1;
   if(nb < 10) return false;
   int kh = iHighest(_Symbol, tf, MODE_HIGH, nb, 0), kl = iLowest(_Symbol, tf, MODE_LOW, nb, 0);
   if(kh < 0 || kl < 0) return false;
   hi = iHigh(_Symbol, tf, kh);
   lo = iLow(_Symbol, tf, kl);
   for(int i = OrdersTotal() - 1; i >= 0; i--)
     {
      if(!OrderSelect(i, SELECT_BY_POS, MODE_TRADES)) continue;
      if(!MyOrder()) continue;
      long key = OrderKey();
      int dir = (OrderType() == OP_BUY) ? 1 : -1;
      double en = OrderOpenPrice(), R = PGet(key, "R", 0);
      double lv[4];
      lv[0] = en;
      lv[1] = PGet(key, "SL0", OrderStopLoss());
      lv[2] = OrderStopLoss();
      lv[3] = (R > 0) ? BoxTarget(key, dir, en, R) : 0;
      for(int j = 0; j < 4; j++) if(lv[j] > 0) { hi = MathMax(hi, lv[j]); lo = MathMin(lo, lv[j]); }
     }
   if(x1 > 0) { hi = MathMax(hi, x1); lo = MathMin(lo, x1); }
   if(x2 > 0) { hi = MathMax(hi, x2); lo = MathMin(lo, x2); }
   if(hi <= lo) return false;
   double pad = 0.07 * (hi - lo);
   hi += pad;
   lo -= pad;
   return true;
  }

string ShotChart(string tag, double x1, double x2, string fileName = "")
  {
   string file = (fileName != "") ? fileName : "AK10_" + _Symbol + "_" + tag + ".png";
   PanelHide();
   double hi = 0, lo = 0;
   bool fix = PhotoRange(x1, x2, hi, lo);
   long oldFix = 0;
   double oldMax = 0, oldMin = 0;
   if(fix)
     {
      //--- fixed scale only for the photo (the old chart setting comes back right after)
      oldFix = ChartGetInteger(0, CHART_SCALEFIX, 0);
      oldMax = ChartGetDouble(0, CHART_FIXED_MAX, 0);
      oldMin = ChartGetDouble(0, CHART_FIXED_MIN, 0);
      ChartSetInteger(0, CHART_SCALEFIX, 1);
      ChartSetDouble(0, CHART_FIXED_MAX, hi);
      ChartSetDouble(0, CHART_FIXED_MIN, lo);
      g_lbPpx = (hi - lo) / (double)ChartPhotoHeight;
     }
   DrawChart();
   ChartRedraw(0);
   bool ok = ChartScreenShot(0, file, ChartPhotoWidth, ChartPhotoHeight, ALIGN_RIGHT);
   if(fix)
     {
      g_lbPpx = 0;
      ChartSetInteger(0, CHART_SCALEFIX, oldFix);
      if(oldFix != 0) { ChartSetDouble(0, CHART_FIXED_MAX, oldMax); ChartSetDouble(0, CHART_FIXED_MIN, oldMin); }
      ChartRedraw(0);
     }
   g_lastPanel = 0;
   UpdatePanel();
   if(!ok || !FileIsExist(file)) return "";
   return file;
  }

//--- one picture to admin / VIP / public, each chat only once per key, as a reply to the trade's entry message
bool SendPhotoOnce(string key, string file, string capV, string capP, bool vip, bool pub, long tradeKey)

      if(id > 0) GlobalVariableSet(gv, (double)TimeCurrent());
      else all = false;
     }
   return all;
  }

//--- content slots: spacing, daily cap and quiet time after signals (shared by all charts)
bool ContentSlotFree(bool timeCritical)
  {
   if(!UseContentManager) return true;
   datetime now = TimeLocal();
   double gap = timeCritical ? 10 : ContentMinGapMin;
   if(now - (datetime)GVGet(TgGV("LASTX"), 0) < gap * 60) return false;
   if(now - (datetime)GVGet(TgGV("LASTPUB"), 0) < (timeCritical ? 5 : QuietAfterSignalMin) * 60) return false;
   if(ContentDailyCap > 0 && GVGet(TgGV("CNT" + DateKey()), 0) >= ContentDailyCap && !timeCritical) return false;
   return true;
  }
void ContentMark() { GVSet(TgGV("LASTX"), (double)TimeLocal()); TgInc("CNT" + DateKey()); }

//--- content posts only in the day hours of the audience (ContentHoursIST, e.g. "9-23"); signals and trade updates: any time
bool ContentHoursNow()
  {
   string s = TrimStr(ContentHoursIST);
   int p = StringFind(s, "-");
   if(p <= 0) return true;
   int a = (int)StringToInteger(StringSubstr(s, 0, p)), b = (int)StringToInteger(StringSubstr(s, p + 1));
   if(a == b || a < 0 || b < 0 || a > 24 || b > 24) return true;
   int h = TimeHour(NowIST());
   return (a < b) ? (h >= a && h < b) : (h >= a || h < b);
  }

// CLOSED-TRADE STATS from the CSV logs of all charts
//------------------------------------------------------------------
int      g_csN = 0, g_csW = 0, g_csL = 0, g_csBE = 0;
double   g_csNet = 0, g_csSumR = 0, g_csBestR = 0;
string   g_csBest = "", g_csSyms = "";
double   g_csDay[40];                     // net USD per IST day (index 0 = oldest of the last 40 days)

string IstText(datetime ist) { return TimeToString(ist, TIME_DATE | TIME_MINUTES); }

void ClosedStats(datetime fromIST)
  {
   g_csN = 0; g_csW = 0; g_csL = 0; g_csBE = 0; g_csNet = 0; g_csSumR = 0; g_csBestR = -999; g_csBest = ""; g_csSyms = "";
   for(int d = 0; d < 40; d++) g_csDay[d] = 0;
   string from = IstText(fromIST);
   datetime today0 = NowIST() - (NowIST() % 86400);
   string syms[];
   int scn[];
   double sr[];
   int ns = 0;
   string fname = "";
   long fh = FileFindFirst("AK10_trades_*.csv", fname);
   if(fh == INVALID_HANDLE) return;
   do
     {
      int h = FileOpen(fname, FILE_READ | FILE_TXT | FILE_ANSI | FILE_SHARE_READ | FILE_SHARE_WRITE);
      if(h == INVALID_HANDLE) continue;
      while(!FileIsEnding(h))
        {
         string ln = FileReadString(h);
         if(StringLen(ln) < 20 || StringFind(ln, "Date,") == 0) continue;
         string p[];
         if(StringSplit(ln, ',', p) < 15) continue;
         string ct = p[11];
         datetime cti = StringToTime(ct);
         int di = 39 - (int)((today0 - (cti - (cti % 86400))) / 86400);
         double pr = StringToDouble(p[13]), rr = StringToDouble(p[14]);
         if(di >= 0 && di < 40) g_csDay[di] += pr;
         if(ct < from) continue;
         g_csN++;
         g_csNet += pr;
         g_csSumR += rr;
         if(rr > 0.05) g_csW++;
         else if(rr < -0.05) g_csL++;
         else g_csBE++;
         if(rr > g_csBestR) { g_csBestR = rr; g_csBest = p[1] + " " + p[2]; }
         int k = -1;
         for(int q = 0; q < ns; q++) if(syms[q] == p[1]) { k = q; break; }
         if(k < 0) { ns++; ArrayResize(syms, ns); ArrayResize(scn, ns); ArrayResize(sr, ns); k = ns - 1; syms[k] = p[1]; scn[k] = 0; sr[k] = 0; }
         scn[k]++;
         sr[k] += rr;
        }
      FileClose(h);
     }
   while(FileFindNext(fh, fname));
   FileFindClose(fh);
   for(int q2 = 0; q2 < ns; q2++) g_csSyms += "\n" + SymEmoji(syms[q2]) + " " + HashTag(syms[q2]) + ": " + IntegerToString(scn[q2]) + " trades, " + Rx(sr[q2]);
  }

//--- equity curve of the last n days as a little bar chart (from the daily results)
string Sparkline(int n)
  {
   if(n > 40) n = 40;
   double c[];
   ArrayResize(c, n);
   double run = 0, mn = 0, mx = 0;
   for(int i = 0; i < n; i++)
     {
      run += g_csDay[40 - n + i];
      c[i] = run;
      if(i == 0 || run < mn) mn = run;
      if(i == 0 || run > mx) mx = run;
     }
   string s = "";
   for(int j = 0; j < n; j++)
     {
      int lv = (mx > mn) ? (int)MathRound(7.0 * (c[j] - mn) / (mx - mn)) : 3;
      s += ShortToString((ushort)(0x2581 + lv));
     }
   return s;
  }

  {
   int wr = (g_csW + g_csL > 0) ? (int)MathRound(100.0 * g_csW / (g_csW + g_csL)) : 0;

              "\n" + Emo(0x1F3AF) + " Win rate (wins vs losses): <b>" + IntegerToString(wr) + "%</b>  " + Bar((int)MathRound(wr / 10.0), 10) +

   return t;
  }

//--- all-time scoreboard per symbol and method (admin)
string ScoreboardText(bool html)
  {
   string pre = g_SP + "SB_";
   string out = "";
   int n = GlobalVariablesTotal();
   for(int i = 0; i < n; i++)
     {
      string nm = GlobalVariableName(i);
      if(StringFind(nm, pre) != 0 || StringSubstr(nm, StringLen(nm) - 2) != "_N") continue;
      string mid = StringSubstr(nm, StringLen(pre), StringLen(nm) - StringLen(pre) - 2);
      double cnt = GlobalVariableGet(nm);
      if(cnt <= 0) continue;
      double w = GVGet(pre + mid + "_W", 0), pl = GVGet(pre + mid + "_PL", 0), r = GVGet(pre + mid + "_R", 0);
      string ln = mid + ": " + DoubleToString(cnt, 0) + " trades, win " + DoubleToString(100.0 * w / cnt, 0) + "%, " + Usd(pl) + " USD, avg " + DoubleToString(r / cnt, 2) + "R";
      out += "\n" + (html ? HtmlEsc(ln) : ln);
     }
   return (out != "") ? "\n\n<b>All time (symbol_method)</b>" + out : "";
  }

string StatusText()
  {
   string s = Emo(0x1F4CB) + " <b>OPEN TRADES</b>\n" + Divider();
   int c = 0;
   double tot = 0;
   for(int i = OrdersTotal() - 1; i >= 0; i--)
     {
      if(!OrderSelect(i, SELECT_BY_POS, MODE_TRADES)) continue;
      if(!IsAKMagic(OrderMagicNumber()) || OrderType() > OP_SELL) continue;
      double pl = OrderProfit() + OrderSwap() + OrderCommission();
      tot += pl;
      string cm = OrderComment();
      StringReplace(cm, TradeComment + " ", "");
      s += "\n" + HtmlEsc(OrderSymbol()) + " " + MethodName(MethodOfMagic(OrderMagicNumber())) + " " + ((OrderType() == OP_BUY) ? "BUY" : "SELL") + " " + DoubleToString(OrderLots(), 2) + " @ " +

      c++;
     }
   if(c == 0) s += "\nNo open trades.";
   else s += "\n\nOpen: " + IntegerToString(c) + " | floating " + Usd(tot) + " USD";
   return s + "\nBalance " + DoubleToString(AccountBalance(), 2) + " | Equity " + DoubleToString(AccountEquity(), 2) + " " + AccountCurrency();
  }

bool IsWeekendIST() { int d = TimeDayOfWeek(NowIST()); return (d == 0 || d == 6); }

string TodayNewsText()
  {
   string out = "";

   int cnt = 0;
   for(int i = 0; i < g_nwN && cnt < 6; i++)
     {

//--- symbols of the charts that are running (registry written by every chart)
int RunningSymbols(string &out[])
  {
   ArrayResize(out, 0);
   int n = GlobalVariablesTotal(), k = 0;
   string pre = TgPreviewOn() ? "AK10QR_" : "AK10R_";
   for(int i = 0; i < n; i++)
     {
      string nm = GlobalVariableName(i);
      if(StringFind(nm, pre) != 0) continue;
      if(TimeLocal() - (datetime)GlobalVariableGet(nm) > 180) continue;
      ArrayResize(out, k + 1);
      out[k++] = StringSubstr(nm, StringLen(pre));
     }
   return k;
  }

string KeyLevelsText()
  {
   string syms[];
   int n = RunningSymbols(syms);
   string out = "";
   bool wk = IsWeekendIST();
   for(int i = 0; i < n; i++)
     {
      string sy = syms[i];
      int cls = ClassifySymbol(sy);
      if(wk && cls != SC_BTC && cls != SC_ETH) continue;
      int dg = (int)MarketInfo(sy, MODE_DIGITS);
      double ph = iHigh(sy, PERIOD_D1, 1), pl = iLow(sy, PERIOD_D1, 1);
      if(ph <= 0 || pl <= 0) continue;
      int kh = iHighest(sy, PERIOD_H4, MODE_HIGH, 20, 1), kl = iLowest(sy, PERIOD_H4, MODE_LOW, 20, 1);

      out += "\n" + SymEmoji(sy) + " <b>" + HtmlEsc(sy) + "</b>  Yesterday H <code>" + DoubleToString(ph, dg) + "</code> L <code>" + DoubleToString(pl, dg) + "</code>" + h4;
     }
   return out;
  }

bool ElectMaster()
  {
   if(!g_tg) return false;
   string n = "AK10TG_M", h = "AK10TG_H";
   double me = (double)ChartID();
   double cur = GVGet(n, 0);
   datetime hb = (datetime)GVGet(h, 0);
   if(cur == me || cur == 0 || TimeLocal() - hb > 30)
     {
      if(cur != me)
        {
         GlobalVariableSet(n, me);
         GlobalVariableSet(h, (double)TimeLocal());
         Sleep(200);
         if(GVGet(n, 0) != me) return false;
         Print("[TG] This chart (", _Symbol, ") now sends the reports / promos and reads the Telegram commands.");
        }
      GlobalVariableSet(h, (double)TimeLocal());
      return true;
     }
   return false;
  }

   if(off > 0) url += "&offset=" + DoubleToString(off, 0);
   string resp = "";
   int code = HttpRequest("GET", url, "", resp);
   if(code != 200)
     {
      if(code == 409)
        {
         //--- another program / webhook reads this bot: sending still works; the webhook is NEVER deleted
         GVSet("AK10TG_BO", (double)(TimeLocal() + 600));
         if(TimeLocal() - (datetime)GVGet("AK10TG_E409", 0) >= 86400)
           {
            GVSet("AK10TG_E409", (double)TimeLocal());
            Print("[TG] 409: another program or webhook reads this bot - commands are handled there. Messages still work.");
           }
        }
      return;
     }
   string marker = "{\"update_id\":";
   int ml = StringLen(marker);
   int pos = 0;
   while(true)
     {
      int p = StringFind(resp, marker, pos);
      if(p < 0) break;
      int nx = StringFind(resp, marker, p + ml);
      string seg = (nx > 0) ? StringSubstr(resp, p, nx - p) : StringSubstr(resp, p);
      double uid = StringToDouble(JsonValue(seg, "update_id", 0));
      if(uid > 0 && uid + 1 > off)
        {
         off = uid + 1;
         GlobalVariableSet("AK10TG_OFF", off);
         HandleUpdate(seg);
        }
      pos = p + ml;
     }
  }

//--- choppiness of the last 24 completed H1 candles (once per H1 candle)
void ChopCalc()
  {
   datetime b = iTime(_Symbol, PERIOD_H1, 0);
   if(b == g_chBar) return;
   g_chBar = b;
   g_chEff = -1; g_chCross = 0; g_chRel = false;
   if(iBars(_Symbol, PERIOD_H1) < 80) return;
   double path = 0;
   for(int i = 1; i <= 24; i++) path += MathAbs(Cl(PERIOD_H1, i) - Cl(PERIOD_H1, i + 1));
   g_chEff = (path > 0) ? MathAbs(Cl(PERIOD_H1, 1) - Cl(PERIOD_H1, 25)) / path : 1.0;
   int prev = 0;
   for(int k = 25; k >= 1; k--)
     {
      int sd = (Cl(PERIOD_H1, k) > EMAv(PERIOD_H1, 20, k)) ? 1 : -1;
      if(prev != 0 && sd != prev) g_chCross++;
      prev = sd;
     }
   //--- a fresh breakout of the 24-candle range with a strong close ends the choppy mode at once
   for(int k = 1; k <= ChopReleaseBars && !g_chRel; k++)
     {
      double hh = Hi(PERIOD_H1, k + 1), ll = Lo(PERIOD_H1, k + 1);
      for(int j = k + 2; j <= k + 24; j++) { hh = MathMax(hh, Hi(PERIOD_H1, j)); ll = MathMin(ll, Lo(PERIOD_H1, j)); }
      double h = Hi(PERIOD_H1, k), l = Lo(PERIOD_H1, k), c = Cl(PERIOD_H1, k), rg = h - l;
      if(rg <= 0) continue;
      if(c > hh && c >= h - rg / 3.0) g_chRel = true;
      if(c < ll && c <= l + rg / 3.0) g_chRel = true;
     }
  }

bool ChopNow()
  {
   ChopCalc();
   return (g_chEff >= 0 && g_chEff < ChopMaxEff && g_chCross >= ChopMinCross && !g_chRel);
  }

//--- lot after the guard (0 = skip the trade); note says why it changed
double GuardLots(double lots, int d, string &note)
  {
   note = "";
   double f = 1.0;
   bool chop = false, brake = false;
   if(ChopGuardHere() && ChopNow())
     {
      bool strong = ChopFullLotAligned && TrendOf(PERIOD_D1) == d && TrendOf(PERIOD_H4) == d && TrendOf(PERIOD_H1) == d;
      if(!strong) { chop = true; f = MathMin(f, ChopLotFactor); }
     }
   if(LossBrakeHere() && LossStreak() >= LossBrakeAfter) { brake = true; f = MathMin(f, LossBrakeFactor); }
   if(chop && brake && LossBrakeSkipChop) { note = "choppy market and " + IntegerToString(LossStreak()) + " losses in a row"; return 0; }
   if(f >= 1.0) return lots;
   note = chop ? (brake ? "choppy market + losing streak: smaller lot" : "choppy market: smaller lot") : "losing streak: smaller lot";
   return NormLot(lots * f);
  }

//--- the panel is built as a list of rows: kind 0 = text line, 1 = section header, 2 = spacer
string g_prT[];
color  g_prC[];
int    g_prK[];
int    g_prN = 0;
void PRow(int kind, string txt, color c)
  {
   if(kind == 0 && StringLen(txt) > PNL_CH)
     {
      //--- wrap long lines (continuation indented), at most 2 lines per row - the panel stays small
      string rest = txt;
      int line = 0;
      while(StringLen(rest) > 0 && line < 2)
        {
         int w = (line == 0) ? PNL_CH : PNL_CH - 2;
         int cut = StringLen(rest);
         if(cut > w)
           {
            cut = w;
            for(int k = w; k > w / 2; k--) if(StringGetCharacter(rest, k) == ' ') { cut = k; break; }
           }
         string part = StringSubstr(rest, 0, cut);
         rest = TrimStr(StringSubstr(rest, cut));
         if(line == 1 && StringLen(rest) > 0) part = StringSubstr(part, 0, MathMax(0, StringLen(part) - 2)) + "..";
         PRow(3, ((line == 0) ? "" : "  ") + part, c);
         line++;
        }
      return;
     }
   ArrayResize(g_prT, g_prN + 1);
   ArrayResize(g_prC, g_prN + 1);
   ArrayResize(g_prK, g_prN + 1);
   g_prT[g_prN] = txt;
   g_prC[g_prN] = c;
   g_prK[g_prN] = (kind == 3) ? 0 : kind;
   g_prN++;
  }

//--- how long until a UTC time, e.g. "2h 10m"
string Until(datetime ev)
  {
   int s = (int)(ev - NowUTC());                                       // ev is UTC (news list)
   if(s <= 0) return "now";
   int h = s / 3600, m = (s % 3600) / 60;
   return (h > 0) ? IntegerToString(h) + "h " + IntegerToString(m) + "m" : IntegerToString(m) + "m";
  }

//--- a click on a panel button
void PanelClick(string obj)
  {
   if(StringFind(obj, PNL + "B_") != 0) return;
   ObjectSetInteger(0, obj, OBJPROP_STATE, 0);
   if(obj == PNL + "B_MIN") { g_pnMin = !g_pnMin; GVSet(PnGV("MIN"), g_pnMin ? 1 : 0); PanelHide(); }
   else if(obj == PNL + "B_DET") { g_pnDet = !g_pnDet; GVSet(PnGV("DET"), g_pnDet ? 1 : 0); PanelHide(); }
   else if(obj == PNL + "B_PAUSE") { GVSet(PauseGV(_Symbol), SymbolPaused() ? 0 : 1); if(GVGet(PauseGV("ALL"), 0) > 0) GVSet(PauseGV("ALL"), 0); }
   else if(obj == PNL + "B_CLOSE")
     {
      if(TimeLocal() - g_pnCloseArm <= 5)
        {
         g_pnCloseArm = 0;
         for(int i = OrdersTotal() - 1; i >= 0; i--)
           {
            if(!OrderSelect(i, SELECT_BY_POS, MODE_TRADES)) continue;
            if(!MyOrder()) continue;
            int dir = (OrderType() == OP_BUY) ? 1 : -1;
            CloseTrade(OrderTicket(), dir, OrderKey(), CR_MANUAL);
           }
        }
      else g_pnCloseArm = TimeLocal();
     }
   g_lastPanel = 0;
   UpdatePanel();
   ChartRedraw(0);
  }

//--- history key for the panel: the last signal of the method, or (none yet) its main setup in the situation now
int PanelHistKey(int m)
  {
   if(g_noteKey[m] >= 0) return g_noteKey[m];
   int ts = TrendSide();
   if(ts == 0) return -1;
   if(m == M_TR)
     {
      bool nt = true;
      for(int c = 1; c <= TX_N; c++) if(TxActive(c, nt)) return HistKeyNow(8 + c, ts);
      return -1;
     }
   return HistKeyNow((m == M_PB) ? 0 : 6, ts);
  }

//--- first start of this version (the list is empty): the last 3 closed trades of this chart from the account history
void PanelLastFromHistory()
  {
   if(IsTesting() || g_lx[0] != "") return;
   int tk[3];
   datetime tc[3];
   ArrayInitialize(tk, -1);
   ArrayInitialize(tc, 0);
   for(int i = OrdersHistoryTotal() - 1; i >= 0; i--)
     {
      if(!OrderSelect(i, SELECT_BY_POS, MODE_HISTORY)) continue;
      if(OrderSymbol() != _Symbol || OrderType() > OP_SELL || !IsMyMagic(OrderMagicNumber())) continue;
      datetime c = OrderCloseTime();
      for(int j = 0; j < 3; j++)
        {
         if(c <= tc[j]) continue;
         for(int q = 2; q > j; q--) { tk[q] = tk[q - 1]; tc[q] = tc[q - 1]; }
         tk[j] = OrderTicket();
         tc[j] = c;
         break;
        }
     }
   for(int k = 0; k < 3; k++)
     {
      if(tk[k] < 0 || !OrderSelect(tk[k], SELECT_BY_TICKET)) continue;
      double p = OrderProfit() + OrderSwap() + OrderCommission();
      g_lx[k] = IstClock(OrderCloseTime()) + "  " + ((OrderType() == OP_BUY) ? "BUY" : "SELL") + " " + MethodTag(MethodOfMagic(OrderMagicNumber())) + "  " + Money(p) + " USD";
      g_lxR[k] = (p > 0.005) ? 1 : ((p < -0.005) ? -1 : 0);
     }
  }

//--- a closed trade goes into the panel's "last trades" list
void PanelAddExit(int dir, int m, double resR, double profit, datetime ct)
  {
   for(int i = 2; i >= 1; i--) { g_lx[i] = g_lx[i - 1]; g_lxR[i] = g_lxR[i - 1]; }
   g_lx[0] = IstClock(ct) + "  " + SideName(dir) + " " + MethodTag(m) + "  " + ((resR >= 0) ? "+" : "") + DoubleToString(resR, 2) + "R  " + Money(profit);
   g_lxR[0] = (resR > 0.05) ? 1 : ((resR < -0.05) ? -1 : 0);
   SaveLast();
  }

   ObjectSetInteger(0, nm, OBJPROP_STYLE, style);
   ObjectSetString(0, nm, OBJPROP_TEXT, txt);
  }

bool InputsOK(string &why)
  {
   why = "";
   if(MagicBase <= 0 || MagicBase > 2000000000) why = "MagicBase must be between 1 and 2000000000";
   else if(LotMode == AK_LOT_FIXED && (LotGold <= 0 || LotBTC <= 0 || LotETH <= 0 || LotEUR <= 0 || MemoryLot <= 0)) why = "every lot input must be > 0";
   else if(LotMode == AK_LOT_RISK && (RiskPercent <= 0 || MemoryRiskPct <= 0)) why = "RiskPercent and MemoryRiskPct must be > 0";
   else if(UsePartial && (PartialPct <= 0 || PartialPct >= 100 || PartialAtR <= 0)) why = "PartialPct must be between 1 and 99 and PartialAtR > 0";
   else if(NoLossAtR < 0 || TrailStartR < 0 || (TrailStartR > 0 && TrailBehindR <= 0)) why = "NoLossAtR / TrailStartR must be >= 0 and TrailBehindR > 0";
   else if(PointsFlipMargin < 1 || PointsFlipMargin > 10) why = "PointsFlipMargin must be 1-10";
   else if(MinHoldHours < 0 || MaxHoldDays < 0) why = "MinHoldHours and MaxHoldDays must be >= 0";
   else if(MaxTradesPerSymbol < 1 || MaxOpenPullback < 0 || MaxOpenBreakout < 0 || MemoryMaxOpen < 0) why = "MaxTradesPerSymbol must be >= 1 and the method limits >= 0";
   else if(CalmVolRank < 0 || WildVolRank > 1 || CalmVolRank >= WildVolRank) why = "0 <= CalmVolRank < WildVolRank <= 1";
   else if(QuickTargetR <= 0 || QuickNoLossR < 0 || QuickMaxHours < 0) why = "QuickTargetR must be > 0, QuickNoLossR and QuickMaxHours >= 0";
   else if(MemoryMinBalance < 0 || MemoryPauseAfter < 0) why = "MemoryMinBalance and MemoryPauseAfter must be >= 0";
   else if(HistoryMinTrades < 1 || HistoryShowLast < 0 || HistoryShowLast > 5) why = "HistoryMinTrades must be >= 1 and HistoryShowLast 0-5";
   else if(MaxOpenTrend < 0 || TrendRiskPct < 0 || (LotMode == AK_LOT_FIXED && (TrendLotH4Gold <= 0 || TrendLotH4BTC <= 0 || TrendLotH4ETH <= 0 || TrendLotH4EUR <= 0))) why = "MaxOpenTrend / TrendRiskPct must be >= 0 and the Trend lots > 0";
   else if(SetupAlertsPerDay < 0 || PromoIntervalMin < 10 || PatternLifeBars < 1) why = "SetupAlertsPerDay >= 0, PromoIntervalMin >= 10 and PatternLifeBars >= 1";
   return (why == "");
  }

int OnInit()
  {
   g_cls = ClassifySymbol(_Symbol);
   g_h = SymbolCode();
   //--- the Strategy Tester uses its own "AK10Q" names, so a test never touches the live/demo chart memory
   g_G  = (IsTesting() ? "AK10Q_" : "AK10_") + IntegerToString(MagicBase + g_h) + "_";
   g_SP = IsTesting() ? "AK10QS_" : "AK10S_";
   if(IsTesting()) GlobalVariablesDeleteAll("AK10Q");                 // every Strategy Tester run starts clean
   string bad = "";
   if(!InputsOK(bad)) { Print("[AK10] invalid inputs: ", bad); return INIT_PARAMETERS_INCORRECT; }
   CheckAccess();
   if(!IsTesting() && GlobalVariableCheck("AK10_GMTOFF")) { g_gmtOff = (int)GlobalVariableGet("AK10_GMTOFF"); g_gmtOk = true; }
   LearnGmtOffset();
   EIABuild();
   NewsBuild();
   g_feedT = 0;
   g_tg = UseTelegram && !IsTesting() && StringLen(TelegramBotToken) > 20;
   if(UseTelegram && !IsTesting() && !g_tg) Print("[TG] Telegram is off: the bot token input is empty.");
   for(int mi = 0; mi < 4; mi++) { g_note[mi] = ""; g_noteT[mi] = 0; g_noteKey[mi] = -1; }
   for(int li = 0; li < 3; li++) { g_lx[li] = ""; g_lxR[li] = 0; g_ptName[li] = ""; g_ptExp[li] = 0; }
   LoadLast();
   PanelLastFromHistory();
   PanelLoadState();
   DeleteObjects(AK_OBJ + "PT");                                        // pattern labels, signal marks and closed-trade boxes of an earlier run
   DeleteObjects(AK_OBJ + "SG");
   DeleteObjects(AK_OBJ + "CB");
   ResumeAfterRestart();
   for(int si = 0; si < MSHN; si++) g_shOn[si] = false;                // shadow trades start empty (they are not kept over a restart)
   g_m5Bar = 0;
   g_lastH1 = iTime(_Symbol, PERIOD_H1, 0);                            // no late entry on a candle that closed before the start
   g_lastM15 = iTime(_Symbol, PERIOD_M15, 0);
   g_lastM5 = iTime(_Symbol, PERIOD_M5, 0);
   g_lastH4 = iTime(_Symbol, PERIOD_H4, 0);
   g_lastM1 = iTime(_Symbol, PERIOD_M1, 0);
   g_ptsBar = 0;
   PointsUpdate();
   UpdateEquityPeak();
   if(!IsTesting())
     {
      if(!EventSetTimer(1)) Print("[AK10] timer could not start - Telegram, the panel refresh and the live calendar will not work");
      ChartSetInteger(0, CHART_SHIFT, 1);                               // room on the right for the labels
      if(ChartGetDouble(0, CHART_SHIFT_SIZE, 0) < 20) ChartSetDouble(0, CHART_SHIFT_SIZE, 20);
     }
   if(!SymbolSupported()) Print("[AK10] ", _Symbol, ": no new trades - ", SymbolOffText(), ". Open trades are still managed.");
   double lot = TradeLots(0);
   Print("[AK10] ", AK_NAME, " ", AK_VER, " on ", _Symbol, " (", ClassName(g_cls), ") magic Pullback ", MethodMagic(M_PB), " / Breakout ", MethodMagic(M_BO), " / Memory ", MethodMagic(M_MEM), " / Trend ", MethodMagic(M_TR),
         " | trading ", (g_canTrade ? "ON" : "BLOCKED: " + g_block), " | lot ", (RiskLotsHere() ? "risk " + DoubleToString(RiskPctHere(), 2) + "%" : DoubleToString(lot, 2)), " | GMT offset ", GmtOffsetSec() / 3600);
   Print("[AK10] methods: Pullback ", (UsePullback ? "ON" : "OFF"), " | Breakout ", (UseBreakout ? "ON" : "OFF"), " | Trend ", (UseTrendMethod ? "ON" : "OFF"), " (", TxActiveList(), ") | Memory ", MemModeShort(),
         " | per symbol ", MaxTradesPerSymbol, " | account ", MaxOpenAccount);
   Print("[AK10] memory: ", (MemTradesHere() ? IntegerToString(MemCellsHere()) + " learnt setups for this symbol, mode " + MemModeText() : "analysis only here"),
         " | history table: ", HIST_N, " rows (class ", HistSymClass(), ") | mode ", HistModeName(), " | news filter ", (UseNewsFilter ? "ON" : "off"), " | open trades taken over: ", g_pkN);
   return INIT_SUCCEEDED;
  }

void OnTick()
  {
   RefreshRates();
   LearnGmtOffset();
   double spr = Ask - Bid;
   if(spr > 0) g_spreadAvg = (g_spreadAvg <= 0) ? spr : g_spreadAvg * 0.995 + spr * 0.005;
   UpdateEquityPeak();
   datetime b1 = iTime(_Symbol, PERIOD_M1, 0);
   bool newM1 = (b1 != g_lastM1 && b1 > 0);
   if(newM1) g_lastM1 = b1;
   PointsUpdate();
   ManagePositions();
   if(!SymbolPaused()) ScanSetups();
   if(MemoryMode != MEM_OFF && MemTradesHere()) MemShadowTick();
   if(newM1)
     {
      MemM5Analysis();
      SetupWatchTick();
      if(IsTesting() && TesterTelegramLog) { TelegramTick(); InstaState(); }   // tester: no timer - the Telegram preview runs once per minute
     }
   if(TimeLocal() - g_lastPanel >= 1) UpdatePanel();
  }

int g_tmr = 0;
void OnTimer()
  {
   if(TimeLocal() - g_lastPanel >= 1) UpdatePanel();                   // at most once a second (OnTick may have just done it)
   g_tmr++;
   if(g_tmr % 3 != 0) return;
   NewsFeedTick();
   TelegramTick();
   InstaState();
  }

void OnChartEvent(const int id, const long &lparam, const double &dparam, const string &sparam)
  {
   if(id == CHARTEVENT_OBJECT_CLICK) PanelClick(sparam);
  }
